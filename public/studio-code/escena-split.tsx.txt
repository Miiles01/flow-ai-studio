/* ============================================================================
   CÓMO USAR ESTA ESCENA (instrucciones para tu asistente de IA:
   Claude, Codex, Gemini, Cursor, Antigravity o cualquier otro)
   ----------------------------------------------------------------------------
   - Es una ESCENA de Remotion: una parte de un Reel (1080×1920 a 30 fps) sobre el video de
     una creadora hablando a cámara. Funciona tal cual: el video, las fotos y las tipografías
     de ejemplo cargan desde wearemiiles.com.
   - Para usarla con TU video:
       1. Pon tu video en public/ (p. ej. "mi-video.mp4") y cambia la ruta del video de ejemplo
          (busca "https://wearemiiles.com/studio/componentes/assets/creadora-hook.mp4") por la tuya.
       2. Transcribe el audio CON TIEMPOS POR PALABRA (p. ej. faster-whisper con word_timestamps)
          y reemplaza la lista P (palabra, segundo en que empieza). Ajusta INICIO al segundo
          de tu video donde empieza la escena.
       3. Los momentos de la animación están atados a palabras (ini(i)): revisa que caigan
          en las palabras correctas de tu guion.
   - Reglas: captions en una línea y al ritmo de la voz; nada de texto arriba de y≈300 ni abajo
     de y≈1450 (interfaz de Reels/TikTok).
   ========================================================================== */
/* ============================================================================
   ESCENA · PANTALLA DIVIDIDA "Entra y sale con motion" · Miiles Estudio
   ----------------------------------------------------------------------------
   Las Escenas son PARTES de un video. Esta va después del hook (mismo video, de 4.86 s a 12.95 s).
   Lo importante aquí es la TRANSICIÓN de la pantalla dividida (arriba solo pasa un carrusel de fotos):
   - Empieza a pantalla completa. En "primeros tres segundos" se abre el split:
       · el panel de arriba cae con un spring que se pasa un poco y se asienta;
       · ella baja 3 cuadros después con su propio spring (720 px: su cara queda entera abajo);
       · el carrusel de adentro llega con parallax (se mueve más que el panel) y todo lleva
         desenfoque de movimiento proporcional a la velocidad.
   - Antes del final se cierra, más rápido de lo que abrió: el panel acelera hacia arriba (ease-in) y
     ella regresa con un spring con rebote a pantalla completa.
   - Captions en UNA línea: en el split, caja blanca sobre la línea del corte (y = 960); a pantalla
     completa, texto blanco debajo de su cara.
   Video: public/creadora-hook.mp4. Guion con tiempos por palabra (faster-whisper).
   ========================================================================== */
import React, { useEffect, useState } from "react";
import { AbsoluteFill, Easing, Img, OffthreadVideo, interpolate, spring, staticFile, useCurrentFrame, useVideoConfig, continueRender, delayRender } from "remotion";

// Archivos: si la ruta empieza con http se usa tal cual (los ejemplos viven en wearemiiles.com);
// si es una ruta local (p. ej. "mi-video.mp4"), se busca en la carpeta public/ de tu proyecto.
const archivo = (p: string) => (/^https?:\/\//.test(p) ? p : staticFile(p));


// ---- Base de las escenas Miiles (tipografías, colores y curvas) ----
const EASE = Easing.bezier(0.22, 1, 0.36, 1);
const clamp = { extrapolateLeft: "clamp", extrapolateRight: "clamp" } as const;
const SANS = "UM-Manrope, Helvetica, Arial, sans-serif";
const DISPLAY = "UM-Advercase, UM-Manrope, sans-serif";   // Advercase/Gambarino SOLO en palabras sin acento
const AZUL = "#4059F1";
const CREMA = "#F4F3EF";
const Fuentes: React.FC = () => {
  const [h] = useState(() => delayRender("Fuentes"));
  useEffect(() => {
    Promise.all([
      new FontFace("UM-Manrope", `url(${archivo("https://wearemiiles.com/studio/componentes/assets/Manrope.woff2")})`, { weight: "200 800" }).load(),
      new FontFace("UM-Advercase", `url(${archivo("https://wearemiiles.com/studio/componentes/assets/Gambarino-Regular.woff2")})`).load(),
    ]).then((fs) => { fs.forEach((x) => document.fonts.add(x)); continueRender(h); }).catch(() => continueRender(h));
  }, [h]);
  return null;
};


const FPS = 30;
const INICIO = 4.86;                                                     // segundo del video donde empieza la escena
export const ESCENA_SPLIT_FRAMES = Math.round((12.95 - INICIO) * FPS);   // ≈ 243 cuadros (8.1 s)
const f = (seg: number) => Math.round((seg - INICIO) * FPS);

const P: [string, number][] = [
  ["Y", 4.88], ["la", 5.02], ["respuesta", 5.2], ["está", 5.5], ["en", 5.78], ["los", 5.88], ["primeros", 6.0], ["tres", 6.44], ["segundos", 6.58],
  ["Si", 7.38], ["no", 7.5], ["enganchas", 7.64], ["ahí", 7.94], ["ya", 8.4], ["perdiste", 8.52],
  ["Empieza", 9.42], ["con", 9.62], ["una", 9.78], ["pregunta", 10.08], ["con", 10.7], ["algo", 10.76], ["que", 10.9], ["sorprenda", 11.08],
  ["o", 11.6], ["con", 11.72], ["el", 11.9], ["resultado", 12.08], ["final", 12.36],
];
const ini = (i: number) => f(P[i][1]);
const GRUPOS: [number, number][] = [[0, 2], [3, 5], [6, 8], [9, 12], [13, 14], [15, 18], [19, 22], [23, 27]];

// ---------------------------------------------------------------------------
// La transición
// ---------------------------------------------------------------------------
const ABRE = ini(6) - 4;       // "primeros tres segundos"
const CIERRA = ini(23) - 8;    // antes de "o con el resultado final"
const BAJA = 720;              // px que baja ella: su cara (y≈250–910) queda completa en la mitad de abajo
const PANEL = { damping: 15, stiffness: 120, mass: 0.9 };    // cae y se pasa apenas
const ELLA = { damping: 12, stiffness: 110, mass: 1 };       // un poco más de rebote
const REGRESO = { damping: 11, stiffness: 150, mass: 0.8 };  // al cerrar: rápida, con rebote

/** 0 = pantalla completa, 1 = split. Abre con spring; cierra con ease-in (más rápido). */
const panelEn = (frame: number, fps: number) => {
  const abre = spring({ frame: frame - ABRE, fps, config: PANEL });
  const cierra = interpolate(frame, [CIERRA, CIERRA + 10], [0, 1], { ...clamp, easing: (t) => t * t * t });
  return abre * (1 - cierra);
};
const ellaEn = (frame: number, fps: number) => {
  const baja = spring({ frame: frame - ABRE - 3, fps, config: ELLA });
  const sube = spring({ frame: frame - CIERRA - 2, fps, config: REGRESO });
  // al cerrar, el panel "jala" a ella: su orilla de arriba nunca queda debajo del panel (no se ve ningún hueco)
  return Math.min(baja - sube, (960 / BAJA) * panelEn(frame, fps) + (frame < CIERRA ? 99 : 0));
};
/** desenfoque de movimiento: proporcional a la velocidad (diferencia con el cuadro anterior) */
const borroso = (v: (fr: number) => number, frame: number, escala: number, max: number) => Math.min(max, Math.abs(v(frame) - v(frame - 1)) * escala);

// ---------------------------------------------------------------------------
// VARIANTES de la transición. Cada una define: qué tan abierto está el split (0–1, decide el estilo de
// los captions), cuánto baja ella (0–1 de BAJA) y cómo se dibuja el panel de arriba.
// El fondo detrás de todo es crema: si en algún momento ella ya bajó y el panel aún no cubre, se ve crema
// (el mismo color del panel), nunca un hueco negro.
// ---------------------------------------------------------------------------
export type VarianteSplit = "rebote" | "suave" | "tiras" | "tarjeta" | "linea";
const ENTRADA_SUAVE = Easing.bezier(0.45, 0, 0.2, 1);          // lenta y sin rebote
const ent = (frame: number, desde: number, dur: number, curva = ENTRADA_SUAVE) => interpolate(frame, [desde, desde + dur], [0, 1], { ...clamp, easing: curva });
const sal = (frame: number, desde: number, dur: number, curva = Easing.bezier(0.55, 0, 0.75, 0.2)) => interpolate(frame, [desde, desde + dur], [0, 1], { ...clamp, easing: curva });

const abiertoV = (v: VarianteSplit, frame: number, fps: number): number => {
  switch (v) {
    case "rebote": return panelEn(frame, fps);
    case "suave": return ent(frame, ABRE - 6, 30) * (1 - ent(frame, CIERRA - 6, 26));
    case "tiras": return ent(frame, ABRE, 22) * (1 - sal(frame, CIERRA, 18));
    case "tarjeta": return spring({ frame: frame - ABRE + 1, fps, config: { damping: 14, stiffness: 110, mass: 0.9 } }) * (1 - sal(frame, CIERRA, 14));
    case "linea": return ent(frame, ABRE + 6, 18, EASE) * (1 - sal(frame, CIERRA + 4, 14));
  }
};
const ellaV = (v: VarianteSplit, frame: number, fps: number): number => {
  switch (v) {
    case "rebote": return ellaEn(frame, fps);
    case "suave": return abiertoV("suave", frame, fps);                       // panel y ella juntos
    case "tiras": return ent(frame, ABRE, 20, EASE) * (1 - sal(frame, CIERRA + 6, 16, ENTRADA_SUAVE));
    case "tarjeta": return ent(frame, ABRE - 2, 18, EASE) * (1 - ent(frame, CIERRA + 12, 16, EASE));   // sube cuando la tarjeta ya se encogió
    case "linea": return ent(frame, ABRE + 6, 20, EASE) * (1 - ent(frame, CIERRA, 18, EASE));
  }
};

/** Panel de arriba según la variante (contenido = carrusel). */
const PanelV: React.FC<{ v: VarianteSplit; children: React.ReactNode }> = ({ v, children }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const a = abiertoV(v, frame, fps);
  if (v === "rebote" || v === "suave") {
    const blur = v === "rebote" ? borroso((fr) => panelEn(fr, fps), frame, 60, 7) : 0;
    if (a <= 0.001) return null;
    return (
      <div style={{ position: "absolute", left: 0, top: -120 - 960 * (1 - a), width: 1080, height: 1080, overflow: "hidden", background: CREMA, filter: blur > 0.3 ? `blur(${blur}px)` : undefined }}>
        <div style={{ position: "absolute", left: 0, top: 120, width: 1080, height: 960, transform: v === "suave" ? `scale(${1.08 - 0.08 * a})` : undefined }}>{children}</div>
      </div>
    );
  }
  if (v === "tiras") {
    // 4 tiras verticales: caen una tras otra (izq → der) y suben en orden inverso
    const N = 4, W = 1080 / N;
    return (
      <>
        {Array.from({ length: N }, (_, k) => {
          const baja = spring({ frame: frame - ABRE - k * 3, fps, config: { damping: 15, stiffness: 140, mass: 0.8 } });
          const sube = sal(frame, CIERRA + (N - 1 - k) * 3, 12);
          const p = baja * (1 - sube);
          if (p <= 0.001) return null;
          return (
            <div key={k} style={{ position: "absolute", left: k * W, top: -960 * (1 - p), width: W + 1, height: 960, overflow: "hidden" }}>
              <div style={{ position: "absolute", left: -k * W, top: 0, width: 1080, height: 960 }}>{children}</div>
            </div>
          );
        })}
      </>
    );
  }
  if (v === "tarjeta") {
    // nace como una pastilla al centro de la mitad de arriba y crece hasta llenarla; al salir se encoge
    if (a <= 0.001) return null;
    const ancho = interpolate(a, [0, 1], [260, 1080]), alto = interpolate(a, [0, 1], [120, 960]);
    const radio = interpolate(a, [0, 0.85, 1], [60, 48, 0], clamp);
    return (
      <div style={{ position: "absolute", left: (1080 - ancho) / 2, top: (960 - alto) / 2, width: ancho, height: alto, borderRadius: radio, overflow: "hidden", background: CREMA }}>
        <div style={{ position: "absolute", left: -(1080 - ancho) / 2, top: -(960 - alto) / 2, width: 1080, height: 960 }}>{children}</div>
      </div>
    );
  }
  // linea: se dibuja una línea blanca en el corte y desde ahí el panel se despliega hacia arriba
  const linea = ent(frame, ABRE - 4, 10, EASE) * (1 - sal(frame, CIERRA + 14, 8));
  const tapa = 960 * (1 - a);                                                       // lo que falta por desplegar (desde arriba)
  return (
    <>
      {a > 0.001 && (
        <div style={{ position: "absolute", left: 0, top: 0, width: 1080, height: 960, overflow: "hidden", clipPath: `inset(${tapa}px 0 0 0)`, background: CREMA }}>
          <div style={{ position: "absolute", inset: 0, transform: `translateY(${(1 - a) * 120}px)` }}>{children}</div>
        </div>
      )}
      {linea > 0.001 && <div style={{ position: "absolute", left: 540 * (1 - linea), width: 1080 * linea, top: 957 - (1 - a) * 0, height: 6, background: "#FFFFFF", borderRadius: 3 }} />}
    </>
  );
};

// ---------------------------------------------------------------------------
// Arriba: solo un carrusel de fotos que pasa de lado
// ---------------------------------------------------------------------------
const FOTOS = ["https://wearemiiles.com/studio/componentes/assets/foto.jpg", "https://wearemiiles.com/studio/componentes/assets/rosa.jpg", "https://wearemiiles.com/studio/componentes/assets/gorra.jpg", "https://wearemiiles.com/studio/componentes/assets/azul.jpg"];
const ANCHO = 500, ALTO = 700, GAP = 36, VEL = 5;   // px por cuadro
const Carrusel: React.FC<{ parallax: number }> = ({ parallax }) => {
  const frame = useCurrentFrame();
  const paso = ANCHO + GAP, total = paso * FOTOS.length;
  const x = -((frame * VEL) % total);
  const tarjetas = [...FOTOS, ...FOTOS, ...FOTOS];
  return (
    <AbsoluteFill style={{ background: CREMA }}>
      <div style={{ position: "absolute", left: 0, top: (960 - ALTO) / 2, height: ALTO, transform: `translate(${x - paso / 2}px, ${parallax}px)` }}>
        {tarjetas.map((src, k) => (
          <div key={k} style={{ position: "absolute", left: k * paso, top: 0, width: ANCHO, height: ALTO, borderRadius: 34, overflow: "hidden" }}>
            <Img src={archivo(src)} style={{ width: "100%", height: "100%", objectFit: "cover" }} />
          </div>
        ))}
      </div>
    </AbsoluteFill>
  );
};

// ---------------------------------------------------------------------------
// Captions (una línea): caja blanca en el corte cuando hay split; blanco debajo de su cara si no
// ---------------------------------------------------------------------------
const Palabra: React.FC<{ i: number }> = ({ i }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const p = spring({ frame: frame - ini(i), fps, config: { damping: 13, stiffness: 190, mass: 0.6 } });
  return (
    <span style={{ display: "inline-block", marginRight: "0.26em", opacity: frame < ini(i) ? 0 : Math.min(1, p * 1.4), transform: `translateY(${(1 - p) * 20}px) scale(${0.86 + 0.14 * p})` }}>{P[i][0]}</span>
  );
};
const Captions: React.FC<{ v: VarianteSplit }> = ({ v }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  return (
    <>
      {GRUPOS.map(([a, b], k) => {
        const desde = ini(a);
        const hasta = k + 1 < GRUPOS.length ? ini(GRUPOS[k + 1][0]) : ESCENA_SPLIT_FRAMES;
        if (frame < desde || frame >= hasta) return null;
        // el estilo lo decide si el split está abierto cuando empieza el grupo (no cambia a la mitad)
        const enSplit = abiertoV(v, desde + 6, fps) > 0.5 && desde + 6 < CIERRA;
        if (enSplit && frame >= CIERRA + 4) return null;                 // se va junto con el panel
        const fin = enSplit ? Math.min(hasta, CIERRA + 4) : hasta;
        const entra = spring({ frame: frame - desde, fps, config: { damping: 15, stiffness: 200, mass: 0.6 } });
        const sale = interpolate(frame, [fin - 4, fin], [1, 0], clamp);
        const idx = Array.from({ length: b - a + 1 }, (_, j) => a + j);
        return (
          <div key={k} style={{ position: "absolute", left: 0, right: 0, top: enSplit ? 960 : 1130, display: "flex", justifyContent: "center", transform: enSplit ? `translateY(-50%) scale(${0.9 + 0.1 * entra})` : undefined, opacity: sale }}>
            <div style={{ background: enSplit ? "#FFFFFF" : "transparent", color: enSplit ? "#111111" : "#FFFFFF", textShadow: enSplit ? "none" : "0 2px 18px rgba(0,0,0,.35)", fontFamily: SANS, fontWeight: 800, fontSize: enSplit ? 56 : 66, letterSpacing: "-0.035em", lineHeight: 1.1, padding: enSplit ? "18px 32px 20px" : 0, borderRadius: 16, whiteSpace: "nowrap" }}>
              {idx.map((i) => <Palabra key={i} i={i} />)}
            </div>
          </div>
        );
      })}
    </>
  );
};

// ---------------------------------------------------------------------------
export const EscenaSplit: React.FC<{ variante?: VarianteSplit }> = ({ variante = "rebote" }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const v = variante;
  const ella = ellaV(v, frame, fps);
  const blurElla = v === "rebote" ? borroso((fr) => ellaEn(fr, fps), frame, 45, 5) : 0;
  const escala = 1 + 0.03 * EASE(interpolate(frame, [0, ESCENA_SPLIT_FRAMES], [0, 1], clamp));
  const parallax = v === "rebote" ? (1 - Math.min(1, panelEn(frame, fps))) * -220 : 0;   // el carrusel llega un poco después que el panel
  return (
    <AbsoluteFill style={{ background: CREMA, overflow: "hidden" }}>
      <Fuentes />
      {/* ella (con audio; la galería lo exporta sin audio) */}
      <AbsoluteFill style={{ transform: `translateY(${BAJA * ella}px) scale(${escala})`, transformOrigin: "50% 30%", filter: blurElla > 0.3 ? `blur(${blurElla}px)` : undefined }}>
        <OffthreadVideo src={archivo("https://wearemiiles.com/studio/componentes/assets/creadora-hook.mp4")} trimBefore={Math.round(INICIO * FPS)} style={{ width: "100%", height: "100%", objectFit: "cover" }} />
      </AbsoluteFill>
      <PanelV v={v}><Carrusel parallax={parallax} /></PanelV>
      <Captions v={v} />
    </AbsoluteFill>
  );
};

export const EscenaSplitSuave: React.FC = () => <EscenaSplit variante="suave" />;
export const EscenaSplitTiras: React.FC = () => <EscenaSplit variante="tiras" />;
export const EscenaSplitTarjeta: React.FC = () => <EscenaSplit variante="tarjeta" />;
export const EscenaSplitLinea: React.FC = () => <EscenaSplit variante="linea" />;
