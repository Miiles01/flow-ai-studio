/* ============================================================================
   CÓMO USAR ESTE COMPONENTE (instrucciones para tu asistente de IA:
   Claude, Codex, Gemini, Cursor, Antigravity o cualquier otro)
   ----------------------------------------------------------------------------
   - Componente de Remotion (React) para video vertical 1080×1920 a 30 fps.
     Si el proyecto no tiene Remotion, créalo (npx create-video@latest), copia este
     archivo en src/ y registra una <Composition> de underlineWordsDuration(props) cuadros.
   - Una frase al centro; las palabras marcadas entre *asteriscos* se subrayan una tras
     otra con el estilo que elijas. Al final se quitan los subrayados (sirve como bucle).
   - Ajustes:
     · text: la frase; líneas con "\n" y palabras a subrayar entre *asteriscos*
       (p. ej. "Contenido que\n*conecta* con\ntu *gente*")
     · underline (estilo):
         "line"     una línea entra de izquierda a derecha y sale por la derecha
         "double"   dos líneas que se cruzan en direcciones opuestas
         "strike"   la línea tacha la palabra por el centro y la palabra crece un poco
         "arc"      un trazo curvo a mano por debajo
         "scribble" un garabato ondulado por debajo
         "wave"     una línea con una onda que pasa por debajo
     · color (texto), lineColor (subrayado, azul Miiles por defecto), background (blanco)
     · font: archivo en public/ (Manrope.woff2 por defecto) · fontSize · weight
     · every: cuadros entre una palabra y la siguiente · hold: pausa con todo subrayado
   - El texto se queda dentro de la zona segura de TikTok / Reels / Shorts.
   - Para renderizar: npx remotion render src/index.ts <Id> out/video.mp4
   ========================================================================== */
// Palabras subrayadas (Remotion) · Miiles Estudio · inspirado en "Line Hover Styles" de Codrops
import React, { useEffect, useState } from "react";
import { AbsoluteFill, Easing, continueRender, delayRender, staticFile, useCurrentFrame, useVideoConfig } from "remotion";

// Archivos: si la ruta empieza con http se usa tal cual (los ejemplos viven en wearemiiles.com y funcionan sin copiar nada);
// si es una ruta local (p. ej. "fotos/mia.jpg"), se busca en la carpeta public/ de tu proyecto.
const archivo = (p: string) => (/^https?:\/\//.test(p) ? p : staticFile(p));


export type UnderlineStyle = "line" | "double" | "strike" | "arc" | "scribble" | "wave";
export type UnderlineWordsProps = {
  text: string;
  underline?: UnderlineStyle;
  color?: string;
  lineColor?: string;
  background?: string;
  font?: string;
  fontSize?: number;
  weight?: number;
  every?: number;
  hold?: number;
};

const clamp = (v: number) => Math.max(0, Math.min(1, v));
const IN = Easing.bezier(0.4, 1, 0.8, 1);      // al subrayar (curva del original)
const OUT = Easing.bezier(0.7, 0, 0.2, 1);     // al quitarlo
const DRAW_IN = Easing.bezier(0.8, 1, 0.7, 1); // trazos dibujados
const DRAW_OUT = Easing.bezier(0.7, 0, 0.3, 1);
const SLIDE = Easing.bezier(0, 0.25, 0.5, 1);

const marked = (text: string) => (text.match(/\*[^*]+\*/g) ?? []).length;
export function underlineWordsDuration(p: Partial<UnderlineWordsProps>) {
  const k = Math.max(1, marked(p.text ?? "*Miiles*"));
  return Math.ceil(20 + k * (p.every ?? 22) + (p.hold ?? 60) + 40);
}

const ARC = "M.945.149C12.3 16.142 43.573 22.572 58.785 10.842";
const SCRIBBLE = "M.426 1.973C4.144 1.567 17.77-.514 21.443 1.48 24.296 3.026 24.844 4.627 27.5 7c3.075 2.748 6.642-4.141 10.066-4.688 7.517-1.2 13.237 5.425 17.59 2.745C58.5 3 60.464-1.786 66 2c1.996 1.365 3.174 3.737 5.286 4.41 5.423 1.727 25.34-7.981 29.14-1.294";
const WAVE = "M0,56.5c0,0,298.666,0,399.333,0C448.336,56.5,513.994,46,597,46c77.327,0,135,10.5,200.999,10.5c95.996,0,402.001,0,402.001,0";

export const UnderlineWords: React.FC<UnderlineWordsProps> = ({
  text, underline = "line", color = "#111111", lineColor = "#4059F1", background = "#FFFFFF",
  font = "https://wearemiiles.com/studio/componentes/assets/Manrope.woff2", fontSize = 104, weight = 600, every = 22, hold = 60,
}) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const [handle] = useState(() => delayRender("Cargando tipografía"));
  const isFile = /\.(woff2?|otf|ttf)$/.test(font);
  useEffect(() => {
    if (!isFile) { continueRender(handle); return; }
    new FontFace("UW-Font", `url(${archivo(font)})`).load().then((f) => { document.fonts.add(f); continueRender(handle); }).catch(() => continueRender(handle));
  }, [font, handle, isFile]);
  const family = `${isFile ? "UW-Font" : font}, Helvetica, Arial, sans-serif`;

  const k = Math.max(1, marked(text));
  const outAt = 20 + k * every + hold; // se quitan todos a la vez (en cascada corta)
  const lines = text.split("\n");
  const longest = Math.max(...lines.map((l) => l.replace(/\*/g, "").length));
  const size = Math.min(fontSize, Math.floor(680 / (longest * 0.55)));
  const th = Math.max(4, size * 0.06); // grosor de la línea

  let idx = -1;
  // estado de cada palabra: p = 0…1 entrando, q = 0…1 saliendo (en segundos como el original)
  const state = (i: number, dIn: number, dOut: number) => {
    const start = 20 + i * every;
    const p = clamp((frame - start) / (dIn * fps));
    const q = clamp((frame - outAt - i * 3) / (dOut * fps));
    return { p, q, on: frame >= start };
  };

  const decorate = (word: string, i: number) => {
    const base: React.CSSProperties = { position: "absolute", left: 0, width: "100%", height: th, background: lineColor, pointerEvents: "none" };
    if (underline === "line") {
      const { p, q } = state(i, 0.3, 0.3);
      // entra de izquierda a derecha (origen 0%) y sale encogiéndose hacia la derecha (origen 100%)
      const s = q > 0 ? 1 - OUT(q) : IN(p);
      return { word: <>{word}</>, deco: <span style={{ ...base, top: "100%", transformOrigin: q > 0 ? "100% 50%" : "0% 50%", transform: `scaleX(${s})` }} /> };
    }
    if (underline === "double") {
      const { p, q } = state(i, 0.3, 0.3);
      const a = q > 0 ? 1 - OUT(q) : IN(p);
      const b = q > 0 ? 1 - OUT(q) : IN(clamp((frame - 20 - i * every - 0.1 * fps) / (0.4 * fps))); // la segunda va 0.1 s después
      return {
        word: <>{word}</>,
        deco: <>
          <span style={{ ...base, top: "100%", transformOrigin: q > 0 ? "100% 50%" : "0% 50%", transform: `scaleX(${a})` }} />
          <span style={{ ...base, top: `calc(100% + ${th * 2.2}px)`, transformOrigin: q > 0 ? "0% 50%" : "100% 50%", transform: `scaleX(${b})` }} />
        </>,
      };
    }
    if (underline === "strike") {
      const { p, q } = state(i, 0.3, 0.3);
      const s = q > 0 ? 1 - OUT(q) : IN(p);
      return {
        word: <span style={{ display: "inline-block", transform: `scale(${1 + 0.1 * s})`, margin: `0 ${0.05 * s * word.length * 0.55 + 0.04 * s}em` }}>{word}</span>,
        deco: <span style={{ ...base, top: "52%", left: "-4%", width: "108%", transformOrigin: q > 0 ? "100% 50%" : "0% 50%", transform: `scaleX(${s})` }} />,
      };
    }
    if (underline === "wave") {
      const { p, q } = state(i, 0.7, 0.5);
      // la línea aparece recta, pasa una onda por debajo y termina recta
      const slide = SLIDE(p);
      const op = q > 0 ? 1 - q : clamp(p * 6);
      return {
        word: <>{word}</>,
        deco: (
          <span style={{ position: "absolute", left: 0, width: "100%", top: "62%", height: "0.6em", overflow: "hidden", opacity: op }}>
            <svg width="300%" height="100%" viewBox="0 0 1200 60" preserveAspectRatio="none" style={{ position: "absolute", left: 0, top: 0, transform: `translateX(${-66.6 * slide}%)`, overflow: "visible" }}>
              <path d={WAVE} fill="none" stroke={lineColor} strokeWidth={th} vectorEffect="non-scaling-stroke" />
            </svg>
          </span>
        ),
      };
    }
    // trazos dibujados: arc y scribble
    const isArc = underline === "arc";
    const { p, q } = state(i, isArc ? 0.4 : 0.6, isArc ? 0.3 : 0.4);
    const drawn = q > 0 ? 1 - DRAW_OUT(q) : DRAW_IN(p);
    return {
      word: <>{word}</>,
      deco: (
        <svg viewBox={isArc ? "0 0 59 18" : "0 0 101 9"} preserveAspectRatio="none"
          style={{ position: "absolute", pointerEvents: "none", overflow: "visible", clipPath: `inset(-50% ${(1 - drawn) * 100}% -50% -10%)`, ...(isArc ? { top: "78%", left: "-12%", width: "118%", height: "0.5em" } : { top: "98%", left: 0, width: "100%", height: "0.26em" }) }}>
          {/* se dibuja de izquierda a derecha (recorte que avanza con el trazo) */}
          <path d={isArc ? ARC : SCRIBBLE} fill="none" stroke={lineColor} strokeWidth={th} strokeLinecap="round" vectorEffect="non-scaling-stroke" />
        </svg>
      ),
    };
  };

  return (
    <AbsoluteFill style={{ background, alignItems: "center", justifyContent: "center", padding: "480px 200px" }}>
      <div style={{ textAlign: "center", color, fontFamily: family, fontWeight: weight, fontSize: size, lineHeight: 1.35, letterSpacing: "-0.03em" }}>
        {lines.map((line, li) => (
          <div key={li} style={{ whiteSpace: "nowrap" }}>
            {line.split(/(\*[^*]+\*)/g).filter(Boolean).map((part, pi) => {
              if (!part.startsWith("*")) return <span key={pi}>{part}</span>;
              idx++;
              const { word, deco } = decorate(part.slice(1, -1), idx);
              return <span key={pi} style={{ position: "relative", display: "inline-block" }}>{word}{deco}</span>;
            })}
          </div>
        ))}
      </div>
    </AbsoluteFill>
  );
};
