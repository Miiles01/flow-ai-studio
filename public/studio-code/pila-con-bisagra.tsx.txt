// Pila con bisagra (Remotion) · Miiles Estudio · inspirado en el efecto 074 "Pinned bottom hinge" de Made With GSAP
import React, { useEffect, useState } from "react";
import { AbsoluteFill, Easing, continueRender, delayRender, staticFile, useCurrentFrame, useVideoConfig, Img } from "remotion";

// Archivos: si la ruta empieza con http se usa tal cual (los ejemplos viven en wearemiiles.com y funcionan sin copiar nada);
// si es una ruta local (p. ej. "fotos/mia.jpg"), se busca en la carpeta public/ de tu proyecto.
const archivo = (p: string) => (/^https?:\/\//.test(p) ? p : staticFile(p));


// ---- Tarjetas (foto, o color con ícono SVG) ----
export type Card = { src?: string; color?: string; icon?: string; iconColor?: string };

/** Contenido de una tarjeta: foto, o color sólido con un ícono SVG de una tinta. */
export const CardFace: React.FC<{ item: Card; iconSize?: string }> = ({ item, iconSize = "40%" }) => (
  <>
    {item.src && <Img src={item.src.startsWith("http") ? item.src : archivo(item.src)} style={{ position: "absolute", inset: 0, width: "100%", height: "100%", objectFit: "cover" }} />}
    {item.icon && (
      <div
        style={{
          position: "absolute", left: "50%", top: "50%", width: iconSize, aspectRatio: "1 / 1",
          transform: "translate(-50%, -50%)", background: item.iconColor ?? "#fff",
          WebkitMaskImage: `url(${archivo(item.icon)})`, maskImage: `url(${archivo(item.icon)})`,
          WebkitMaskSize: "contain", maskSize: "contain", WebkitMaskRepeat: "no-repeat", maskRepeat: "no-repeat",
          WebkitMaskPosition: "center", maskPosition: "center",
        }}
      />
    )}
  </>
);

export type HingeStackProps = {
  items: Card[];
  left?: string;
  right?: string;
  background?: string;
  color?: string;
  font?: string;
  cardWidth?: number;
  cardHeight?: number;
  hold?: number;
  flip?: number;
};

export function hingeStackDuration(p: Partial<HingeStackProps>) {
  return (p.items?.length ?? 5) * ((p.hold ?? 14) + (p.flip ?? 52));
}

const BISAGRA = Easing.bezier(0.6, 0, 0.3, 1);
const FLUIDO = Easing.bezier(0.45, 0, 0.25, 1);

export const HingeStack: React.FC<HingeStackProps> = ({
  items, left = "Contenido UGC", right = "Trabajemos juntos", background = "#FFFFFF", color = "#111111",
  font = "https://wearemiiles.com/studio/componentes/assets/Manrope.woff2", cardWidth = 660, cardHeight = 840, hold = 14, flip = 52,
}) => {
  const frame = useCurrentFrame();
  const { width, height } = useVideoConfig();
  const [handle] = useState(() => delayRender("Cargando tipografía"));
  useEffect(() => {
    new FontFace("HS-Font", `url(${archivo(font)})`).load().then((f) => { document.fonts.add(f); continueRender(handle); }).catch(() => continueRender(handle));
  }, [font, handle]);

  const n = items.length;
  const ciclo = hold + flip;
  const top = height / 2 - cardHeight / 2 + 80;
  const leftX = width / 2 - cardWidth / 2;

  // El mazo da vueltas: la tarjeta de enfrente cae sobre su borde de abajo, se va hacia atrás
  // y se acomoda al fondo de la pila mientras las demás avanzan un lugar. Al terminar n vueltas
  // todo queda como al inicio (bucle perfecto).
  const vuelta = Math.floor(frame / ciclo);
  const tRaw = (frame - vuelta * ciclo - hold) / flip;
  const t = Math.max(0, Math.min(1, tRaw));
  const avance = FLUIDO(Math.max(0, Math.min(1, (t - 0.15) / 0.85)));   // las demás avanzan con calma
  const FONDO = 210, VISIBLES = 4;                     // profundidad entre tarjetas

  // todas apoyadas en el mismo piso (el borde de abajo de la de enfrente); las de atrás solo
  // están más lejos, y por la perspectiva (mirando un poco desde arriba) asoman por encima
  const lugar = (k: number) => ({ y: 0, z: -k * FONDO });

  return (
    <AbsoluteFill style={{ background, overflow: "hidden", fontFamily: "HS-Font, Helvetica, Arial, sans-serif", color }}>
      <div style={{ position: "absolute", left: leftX, right: leftX, top: top - 300, display: "flex", justifyContent: "space-between", fontSize: 38, fontWeight: 500, letterSpacing: "-0.02em" }}>
        <span>{left}</span><span>{right}</span>
      </div>
      <div style={{ position: "absolute", inset: 0, perspective: 2000, perspectiveOrigin: `50% ${top - 420}px`, transformStyle: "preserve-3d" }}>
        <div style={{ position: "absolute", inset: 0, transformStyle: "preserve-3d" }}>
          {items.map((frente, i) => {
            const atras = items[(i + 1) % n];
            const k0 = ((i - vuelta) % n + n) % n;          // lugar en la pila (0 = enfrente)
            let ang = 0, pos = lugar(k0), alfa = 1;
            if (k0 === 0) {
              // 1ª mitad: cae hacia nosotros sobre el borde de abajo (se ve la cara de atrás)
              // 2ª mitad: se va hacia el fondo y se levanta otra vez, ya detrás de todas
              // un solo movimiento continuo (sin pausa a la mitad): gira toda la vuelta mientras,
              // desde antes de terminar de caer, ya va viajando hacia el fondo
              const giro = FLUIDO(t);
              const viaje = FLUIDO(Math.max(0, Math.min(1, (t - 0.28) / 0.72)));
              ang = -360 * giro;
              const fin = lugar(n - 1);
              pos = { y: 0, z: fin.z * viaje };                // se desliza hacia atrás pegada al piso
            } else {
              const k = k0 - avance;                        // las demás avanzan un lugar
              pos = lugar(k);
            }
            // las de muy atrás se desvanecen (la que cae se va desvaneciendo si su nuevo lugar queda fuera de vista)
            const kVisual = k0 === 0 ? (n - 1) * FLUIDO(Math.max(0, Math.min(1, (t - 0.28) / 0.72))) : k0 - avance;
            if (kVisual > VISIBLES) alfa = 0;
            else if (kVisual > VISIBLES - 1) alfa = VISIBLES - kVisual;
            return (
              <div key={i} style={{
                position: "absolute", left: leftX, top, width: cardWidth, height: cardHeight,
                transformStyle: "preserve-3d", transformOrigin: "50% 100%",
                transform: `translate3d(0, ${pos.y}px, ${pos.z}px) rotateX(${ang}deg)`,
              }}>
                {/* la opacidad va en cada cara: si va en la tarjeta, el navegador aplana el 3D */}
                <div style={{ position: "absolute", inset: 0, borderRadius: 24, overflow: "hidden", background: frente.color ?? "#EEEFF2", backfaceVisibility: "hidden", opacity: alfa }}>
                  <CardFace item={frente} />
                </div>
                <div style={{ position: "absolute", inset: 0, borderRadius: 24, overflow: "hidden", background: atras.color ?? "#EEEFF2", backfaceVisibility: "hidden", transform: "rotateX(180deg)", opacity: alfa }}>
                  <CardFace item={atras} />
                </div>
              </div>
            );
          })}
        </div>
      </div>
    </AbsoluteFill>
  );
};
